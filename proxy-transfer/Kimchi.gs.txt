/* Code.gs 의 handleChat_ 를 대체한다. app 이 'kimchi' 가 아니면 도랏이 동작과 동일. */
function handleChat_(req) {
  var text = String((req && req.text) || '').slice(0, MAX_CHARS).trim();
  if (!text) return json_({ ok: false, error: 'empty' });
  var nick = String((req && req.nick) || '').slice(0, 12).trim();
  var app = String((req && req.app) || '').trim();
  var kimchi = (app === 'kimchi');
  var tone = String((req && req.tone) || 'formal').trim();
  var casual = kimchi && (tone === 'casual');
  var call = kimchi && !!(req && req.call);
  var ctx = kimchi ? String((req && req.ctx) || '').slice(0, 400).trim() : '';
  /* 통화는 문맥도 더 짧게(2턴) — 페이로드가 곧 대기 시간이다 */
  var histLimit = call ? 4 : MAX_TURNS * 2;
  var history = (req && req.history && req.history.length) ? req.history.slice(-histLimit) : [];
  var key = apiKey_();
  if (!key) return json_({ ok: false, error: 'no_key' });

  var sys = kimchi ? (KIMCHI_SYSTEM + '\n\n' + (casual ? KIMCHI_TONE_CASUAL : KIMCHI_TONE_FORMAL)) : CHAT_SYSTEM;
  if (call) sys += '\n\n' + KIMCHI_CALL;
  if (nick) {
    var callName = casual ? nick.replace(/님$/, '') : (/님$/.test(nick) ? nick : nick + '님');
    sys += '\n\n지금 사용자의 이름: ' + callName + ' (이 사람을 부를 때는 항상 "' + callName + '"이라고 쓴다)';
  }
  if (ctx) sys += '\n\n지금 상황 정보(사실로 취급한다):\n' + ctx;
  var contents = buildContents_(history, text);
  var cfg = {
    temperature: kimchi ? (call ? 0.7 : 0.85) : 0.9,
    topP: 0.95,
    maxOutputTokens: kimchi ? (call ? 140 : 320) : 400
  };

  var want = String((req && req.model) || '').trim();
  var r = runChain_(key, sys, contents, cfg, chatChain_(want), 1, true);
  if (!r.ok) return json_({ ok: false, error: 'upstream', tried: r.tried });
  return json_({ ok: true, reply: r.text, model: r.model, app: kimchi ? 'kimchi' : 'dorati', call: call });
}
